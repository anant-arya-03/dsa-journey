// ======================= 4 SUM =======================
//like 3sum problem i loop n-3 tak chalta tha kyuki last 2 elemet ko jagha deni hai
// Approach:
// 1. Sort the array.
// 2. Fix first element using i.
// 3. Fix second element using j.
// 4. Use two pointers:
//      start = j + 1
//      end = n - 1
//
// Then:
//
// sum == target  -> answer mil gaya
//                   start++ and end--
//
// sum < target   -> sum chhota hai
//                   start++
//
// sum > target   -> sum bada hai
//                   end--
//
// Duplicate values ko skip karna hai so that
// same quadruplet baar-baar na aaye.
//
// Time Complexity: O(n^3)
// Space Complexity: O(1) excluding answer


class Solution {
public:
    vector<vector<int>> fourSum(vector<int>& nums, int target) {

        vector<vector<int>> ans;

        // Array ko sort karenge
        sort(nums.begin(), nums.end());

        int n = nums.size();

        // First element fix
        for (int i = 0; i < n - 3; i++) {

            // Same i ko dobara process nahi karna
            if (i > 0 && nums[i] == nums[i - 1])
                continue;

            // Second element fix
            for (int j = i + 1; j < n - 2; j++) {

                // Same j ko dobara process nahi karna
                if (j > i + 1 && nums[j] == nums[j - 1])
                    continue;

                // Remaining 2 elements ke liye
                // two pointer use karenge
                int start = j + 1;
                int end = n - 1;

                while (start < end) {

                    // 4 numbers ka sum target ke equal hai
                    if ((long long)nums[i] + nums[j]
                        + nums[start] + nums[end] == target) {

                        ans.push_back({
                            nums[i],
                            nums[j],
                            nums[start],
                            nums[end]
                        });

                        // Answer mil gaya,
                        // dono pointers ko move karo
                        start++;
                        end--;

                        // Duplicate start ko skip karo
                        while (start < end &&
                               nums[start] == nums[start - 1])
                            start++;

                        // Duplicate end ko skip karo
                        while (start < end &&
                               nums[end] == nums[end + 1])
                            end--;
                    }

                    // Sum chhota hai
                    // right side jaane ke liye start++
                    else if ((long long)nums[i] + nums[j]
                             + nums[start] + nums[end] < target) {

                        start++;
                    }

                    // Sum bada hai
                    // left side aane ke liye end--
                    else {
                        end--;
                    }
                }
            }
        }

        return ans;
    }
};
