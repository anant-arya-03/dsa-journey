merge_sort


#include <iostream>
#include <vector>
using namespace std;

void merge_array(int arr[], int low, int mid, int high)
{
    int position = 0;
    int size = high - low + 1;
    int *a = new int[size];

    int first = low;
    int second = mid + 1;

    while (first <= mid && second <= high)
    {
        if (arr[first] <= arr[second])
        {
            a[position++] = arr[first++];
        }
        else
        {
            a[position++] = arr[second++];
        }
    }

    while (first <= mid)
    {
        a[position++] = arr[first++];
    }

    while (second <= high)
    {
        a[position++] = arr[second++];
    }

    position = 0;

    for (int i = low; i <= high; i++)
    {
        arr[i] = a[position++];
    }

    delete[] a;
}

void divide_array(int arr[], int low, int high)
{
    if (low >= high)
    {
        return;
    }

    int mid = low + (high - low) / 2;

    divide_array(arr, low, mid);
    divide_array(arr, mid + 1, high);

    merge_array(arr, low, mid, high);
}

int main()
{
    int arr[] = {4, 7, 2, 5, 6, 8};
    int n = sizeof(arr) / sizeof(arr[0]);

    divide_array(arr, 0, n - 1);

    cout << "Sorted array: ";

    for (int i = 0; i < n; i++)
    {
        cout << arr[i] << " ";
    }

    return 0;
}
