
class Solution {
public:
    int climbStairs(int n) {
        if (n == 1) {
            return 1;
        }
        else if (n == 2) {
            return 2;
        }
        long long ans = climbStairs(n - 1) + climbStairs(n - 2);

        return ans;
    }
};
