class Solution {
public:
    int findLengthOfLCIS(vector<int>& nums) {
       int start = 0, end=0, n = nums.size(), longest_substring = 1;
       for(int i =0; i<n-1; i++){
        if(nums[i]< nums[i+1]){
            end++;
        }
        else{
            start++;
            end=start;     
        }


          longest_substring = max(longest_substring, end-start+1);
       } return longest_substring;
    }
};
