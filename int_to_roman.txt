class Solution {
public:

    string ones(int n)
    {
        if(n == 1) return "I";
        if(n == 2) return "II";
        if(n == 3) return "III";
        if(n == 4) return "IV";
        if(n == 5) return "V";
        if(n == 6) return "VI";
        if(n == 7) return "VII";
        if(n == 8) return "VIII";
        if(n == 9) return "IX";

        return "";
    }

    string tens(int n)
    {
        if(n == 1) return "X";
        if(n == 2) return "XX";
        if(n == 3) return "XXX";
        if(n == 4) return "XL";
        if(n == 5) return "L";
        if(n == 6) return "LX";
        if(n == 7) return "LXX";
        if(n == 8) return "LXXX";
        if(n == 9) return "XC";

        return "";
    }

    string hundreds(int n)
    {
        if(n == 1) return "C";
        if(n == 2) return "CC";
        if(n == 3) return "CCC";
        if(n == 4) return "CD";
        if(n == 5) return "D";
        if(n == 6) return "DC";
        if(n == 7) return "DCC";
        if(n == 8) return "DCCC";
        if(n == 9) return "CM";

        return "";
    }

    string thousands(int n)
    {
        if(n == 1) return "M";
        if(n == 2) return "MM";
        if(n == 3) return "MMM";

        return "";
    }

    string intToRoman(int num)
    {
        int thousand = num / 1000;
        int hundred = (num / 100) % 10;
        int ten = (num / 10) % 10;
        int one = num % 10;

        string ans;

        ans += thousands(thousand);
        ans += hundreds(hundred);
        ans += tens(ten);
        ans += ones(one);

        return ans;
    }
};




INTEGER TO ROMAN — SUMMARY

Main Idea:
The number is divided into 4 parts: Thousands → Hundreds → Tens → Ones.
Each part is converted separately using a function, and then all the Roman strings are joined together.

1. ones(n):
Handles the ones digit (1–9):
1→I, 2→II, 3→III, 4→IV, 5→V, 6→VI, 7→VII, 8→VIII, 9→IX.
If n = 0, return "" because there is nothing to add.

2. tens(n):
Handles the tens digit:
1→X, 2→XX, 3→XXX, 4→XL, 5→L, 6→LX, 7→LXX, 8→LXXX, 9→XC.
Here n represents the digit, so n=4 means 40 → XL.

3. hundreds(n):
Handles the hundreds digit:
1→C, 2→CC, 3→CCC, 4→CD, 5→D, 6→DC, 7→DCC, 8→DCCC, 9→CM.

4. thousands(n):
Handles the thousands digit:
1→M, 2→MM, 3→MMM.
The number is at most 3999, so only 1–3 are needed.

Digit Extraction:
int thousand = num / 1000;
int hundred  = (num / 100) % 10;
int ten      = (num / 10) % 10;
int one      = num % 10;

Rules:
 /10   → removes 1 digit from the right
 /100  → removes 2 digits from the right
 /1000 → removes 3 digits from the right
 %10   → extracts the last digit

Example: num = 3689
thousand = 3
hundred  = 6
ten      = 8
one      = 9

Building the Answer:
string ans;
ans += thousands(thousand);
ans += hundreds(hundred);
ans += tens(ten);
ans += ones(one);

For 3689:
thousands(3) → MMM
hundreds(6)  → DC
tens(8)      → LXXX
ones(9)      → IX

Final answer:
MMM + DC + LXXX + IX = MMMDCLXXXIX

KEY IDEA:
First separate the number into place-value digits, then convert each digit into its Roman representation, and finally concatenate them in the order:
Thousands → Hundreds → Tens → Ones.
