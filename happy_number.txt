class Solution {
public:
    int splitNumber(int num) {
        int sum = 0;

        while (num > 0) {
            int remainder = num % 10;
            sum = sum + remainder * remainder;
            num = num / 10;
        }

        return sum;
    }

    bool isHappy(int n) {
        int ans = splitNumber(n);
        int ans2 = splitNumber(ans);

        while (ans != ans2) {
            ans = splitNumber(ans);
            ans2 = splitNumber(splitNumber(ans2));
        }

        if (ans2 == 1) {
            return true;
        } else {
            return false;
        }
    }
};
