
#include <iostream>
using namespace std; 

int maxSubarray(int arr[], int n)
{
    int maxSum = INT_MIN;

    for(int i = 0; i < n; i++)
    {
        for(int j = i; j < n; j++)
        {
            int sum = 0;

            for(int k = i; k <= j; k++)
            {
                sum += arr[k];
            }

            if(sum > maxSum)
                maxSum = sum;
        }
    }

    return maxSum;
}

int main(){
int array[]={1, 2, 3, 4, 5};
int arr_size = sizeof(array)/sizeof(int);
cout<<maxSubarray(array, arr_size)<<endl;
}
