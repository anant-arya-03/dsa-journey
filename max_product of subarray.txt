class Solution {
public:
    int maxProduct(vector<int>& nums) {
        int max_product= INT_MIN;
        for(int i=0;i<nums.size();i++){
        int product =1;
            for(int j=i;j<nums.size();j++){
                product *= nums[j];
                max_product = max(product, max_product);
            }
        }
        return max_product; 
    }
};
